#!/usr/bin/env node
// PreToolUse: blocks writes to protected files, undeclared packages, deferred dependencies and
// hook bypass attempts (exit 2 → the tool call is cancelled and the message goes to the agent).
// Claude Code only cancels a PreToolUse call on exit 2: by default a timed-out hook does not block
// and any other exit code (1, a signal such as a segfault's 139) is a non-blocking error.
// Two layers keep this failing closed (https://code.claude.com/docs/en/hooks):
//  1. settings.json sets `"onFailure": "block"` (Claude Code v2.1.295+), so timeouts and other
//     exit codes block, plus an explicit `timeout` (seconds; command hooks default to 600) above LIMIT_MS.
//     The hook uses the documented exec form (`command: "node"` + `args`, no shell) so it also runs on
//     Windows without Git Bash, where `$CLAUDE_PROJECT_DIR` is not expanded.
//     UNVERIFIED RISK on old versions: the docs do not say which version added `args`, nor whether older
//     versions ignore or reject unknown hook keys (`args`, `onFailure`). If a version ignores `args`, it
//     would run a bare `node` (reads stdin as JS, exits 1 = non-blocking) and this guard would fail open;
//     if it rejects the entry or the file, the guard would not run at all. Use a current Claude Code
//     (v2.1.295+); CI arch-check remains the backstop.
//  2. This file is a supervisor for older versions and for a clear message: it runs the real decision
//     (guard-worker.mjs → scripts/guard-rules.mjs) in a child Node process with a wall-clock limit and a
//     heap cap, and maps every outcome except a clean exit 0 / exit 2 to a block.
// Cost of layer 2: one extra Node start per call (~40 ms), accepted so a native crash cannot fail open.
import { spawnSync } from "node:child_process";
import { readFileSync } from "node:fs";
import { fileURLToPath } from "node:url";

const DEFAULT_LIMIT_MS = 10_000;
// GUARD_TIMEOUT_MS may only lower the limit (stricter), never raise it.
const requested = Number(process.env.GUARD_TIMEOUT_MS);
const LIMIT_MS =
  Number.isInteger(requested) && requested > 0 && requested < DEFAULT_LIMIT_MS ? requested : DEFAULT_LIMIT_MS;
const MAX_HEAP_MB = 512;

const block = (message) => `⛔ ${message}\nRules: CLAUDE.md and architecture.json.\n`;
const failed = (why) =>
  block(`The guard hook failed (${why}); blocking this call. Ask the user to check .claude/hooks/guard.mjs and architecture.json.`);

// Returns the stderr text for a block (exit 2) or null to allow. The caller sets process.exitCode and
// lets the process end naturally: process.exit() can truncate a long message on a piped stderr.
function supervise() {
  let stdin;
  try {
    stdin = readFileSync(0);
  } catch (error) {
    return failed(error instanceof Error ? error.message : String(error));
  }

  const worker = fileURLToPath(new URL("./guard-worker.mjs", import.meta.url));
  const child = spawnSync(process.execPath, [`--max-old-space-size=${MAX_HEAP_MB}`, worker], {
    input: stdin,
    timeout: LIMIT_MS,
    killSignal: "SIGKILL",
    maxBuffer: 1024 * 1024,
    env: process.env,
    stdio: ["pipe", "ignore", "pipe"],
    windowsHide: true,
  });

  if (child.error) {
    return failed(child.error.code === "ETIMEDOUT" ? `timed out after ${LIMIT_MS} ms` : child.error.message);
  }
  if (child.signal) return failed(`worker killed by ${child.signal}`);
  if (child.status === 2) {
    const message = child.stderr.toString();
    return message.trim() ? message : failed("worker blocked the call without giving a reason");
  }
  if (child.status !== 0) return failed(`worker exited with code ${child.status}`);
  return null;
}

const blocked = supervise();
if (blocked !== null) {
  process.stderr.write(blocked);
  process.exitCode = 2;
}

# Project rules

This block is protected by `.claude/hooks/guard.mjs` and `pnpm arch`. Anything outside it may be edited freely.

## Language
Chat with the user in the user's language. Everything else — code, comments, docs, ADRs, commit messages, PR text, log and error messages — is written in English.

## Commands
- `pnpm verify`: architecture, lint, typecheck and tests. **Work is done only when this passes.**
- `pnpm dev`: all apps. Postgres via `docker compose up -d`.
- `pnpm arch`: architecture check only.
- `pnpm worktree:setup` / `worktree:info`: per-worktree `.env` + database (Paseo runs setup automatically; see `paseo.json`). Never hard-code ports; services get `$PASEO_PORT`.
- `pnpm add <pkg> --filter @acme/<target>`: add a dependency; the version is written to the catalog.

## Where code goes
| Need | Location |

| Table, migration, query helper | `packages/db` |
| Use case (domain + db orchestration) | `packages/core` |
| HTTP endpoint | `apps/api/src/app.ts` (validation + core call only) |

| UI | `apps/web` (API access via `src/lib/api.ts`) |

Dependency direction and package roles: `architecture.json` and `docs/adr/0001-package-boundaries.md`.

# 0001. Package boundaries

- Status: accepted

## Decision
The monorepo consists only of the packages listed in `architecture.json`. Dependency direction:

```
web ──(import type only)──▶ api ──▶ core ──▶ db ──▶ domain
 │                           │       └────▶ contracts
 └──▶ contracts, domain      └──▶ contracts
worker ──▶ core, db, domain
```

- `domain` depends on no package and no external library.
- `contracts` depends only on `zod`; validation rules are written only here.
- `api`/`worker` contain no business rules; they call `core` use cases. They import `db` only to wire up the connection (composition root).
- `web` reaches the API only through `hc<AppType>`; it never imports `db`/`core`.
- Relative imports across packages (`../../packages/...`) are forbidden.

source imports, deferred dependencies, `@ts-ignore`/`@ts-nocheck`, empty placeholder files and the

# 0004. Per-worktree isolation (Paseo)

- Status: accepted

## Context
Agents and developers work in parallel git worktrees (Paseo, Claude Code, `git worktree add`).
Sharing one database or fixed ports between them causes collisions.

## Decision
- One shared local Postgres; isolation by name. The main checkout uses `acme`, every
  worktree `acme_wt_<key>` (`<key>` = sanitized folder name + sha1 of the path).
  pg-boss stores its queue in the same database, so jobs are isolated too.
- `scripts/worktree.mjs` (`pnpm worktree:setup|teardown|info|gc`) writes `.env` (copied from the
  source checkout, only isolated keys rewritten, placeholder secrets generated), creates the
  database, runs migrations, and drops it again on teardown. It never touches the main checkout's
  database; teardown always exits 0.
- `paseo.json` runs setup/teardown and defines `api`, `web`, `worker` as services. Ports are
  assigned by Paseo (`servicePorts.range` 4100-4999) and never hard-coded; the web dev server
  proxies `/api` to `API_URL`. `.worktreeinclude` copies `.env` into new worktrees.
- Paseo reads `paseo.json` from the base branch: changes take effect after merging to main.

## Consequences
Worktrees removed outside Paseo leave databases behind; `pnpm worktree:gc --yes` drops them.

/** Money: amount in minor units (cents) as bigint + ISO 4217 currency. Never use number. */
export interface Money {
  readonly amount: bigint;
  readonly currency: string;
}

export class CurrencyMismatchError extends Error {
  override name = "CurrencyMismatchError";
}

const digitsCache = new Map<string, number>();

/** Number of minor-unit digits for a currency (USD: 2, JPY: 0, KWD: 3). */
export function minorDigits(currency: string): number {
  let digits = digitsCache.get(currency);
  if (digits === undefined) {
    digits =
      new Intl.NumberFormat("en", { style: "currency", currency }).resolvedOptions()
        .maximumFractionDigits ?? 2;
    digitsCache.set(currency, digits);
  }
  return digits;
}

export function money(amount: bigint, currency: string): Money {
  const code = currency.toUpperCase();
  minorDigits(code); // throws RangeError for an invalid currency
  return { amount, currency: code };
}

export function add(a: Money, b: Money): Money {
  if (a.currency !== b.currency) {
    throw new CurrencyMismatchError(`Cannot add ${a.currency} and ${b.currency}`);
  }
  return money(a.amount + b.amount, a.currency);
}

/** Lossless decimal representation: 12345n USD → "123.45" */
export function toDecimalString(m: Money): string {
  const digits = minorDigits(m.currency);
  const negative = m.amount < 0n;
  const abs = (negative ? -m.amount : m.amount).toString().padStart(digits + 1, "0");
  const whole = digits === 0 ? abs : abs.slice(0, -digits);
  const fraction = digits === 0 ? "" : `.${abs.slice(-digits)}`;
  return `${negative ? "-" : ""}${whole}${fraction}`;
}

export function format(m: Money, locale: string): string {
  return new Intl.NumberFormat(locale, { style: "currency", currency: m.currency }).format(
    toDecimalString(m) as `${number}`,
  );
}

/** Files and empty folders below a package, in walk order, with the source of code files. */
function entriesOf(dir, out = []) {
  if (!existsSync(dir)) return out;
  for (const e of readdirSync(dir, { withFileTypes: true })) {
    if (e.name === "node_modules" || e.name === "dist" || e.name === ".turbo") continue;
    const p = path.join(dir, e.name);
    if (e.isDirectory()) {
      if (readdirSync(p).length === 0) out.push({ path: rel(p), kind: "emptyDir" });
      entriesOf(p, out);
    } else {
      out.push({ path: rel(p), kind: "file", source: SOURCE_RE.test(p) ? readFileSync(p, "utf8") : undefined });
    }
  }

}

  if (relsOf(g.operands, gb).some((r) => isProtectedOrDir(contract, r) || holdsProtected(contract, r))) return MSG.git;
  // A whole-tree revert from a revision rewrites protected files too: `checkout <rev> -- .`, `restore --source=<rev> .`.
  const dots = g.operands.filter((o) => o.text === "." || o.text === "./" || o.text === ":/");
  const fromRev =
    (sub === "checkout" && g.operands.length > dots.length) ||
    (sub === "restore" && g.flags.some((x) => /^(?:--source|-s)/.test(x)));
  if (dots.length && fromRev) return MSG.git;
  // A glob after a revision restores every path it matches: `checkout HEAD~5 -- '*'`, `restore -s rev '*.json'`.
  const glob = (o) => /[*?[]/.test(o.text);
  const rest = args.slice(i + 1);
  const dd = rest.findIndex((a) => a.text === "--");
  const hasRev =
    sub === "restore"
      ? g.flags.some((x) => /^(?:--source|-s)/.test(x))
      : sub === "checkout" && (dd >= 0 ? rest.slice(0, dd).some((a) => !a.text.startsWith("-")) : g.operands.some((o) => !glob(o)));
  if (hasRev && g.operands.filter(glob).some((o) => relsOf([o], gb).some((r) => globHitsProtected(contract, r)))) return MSG.git;
  return undefined;
}

/** rm / mv of a folder that holds protected paths; `rm -rf .` only when no `cd` moved elsewhere first. */
function removeRule(env, c) {
  const recursive = ["chmod", "chown", "chgrp"].includes(c.name) && c.args.some((a) => /^-[A-Za-z]*R|^--recursive$/.test(a.text));
  if (c.name !== "rm" && c.name !== "mv" && !recursive) return undefined;
  const { flags, operands } = splitArgs(c.args);
  const byTarget = flags.some((x) => /^--target-directory|^-[A-Za-z]*t$/.test(x));
  const sources = c.name !== "mv" || byTarget ? operands : operands.slice(0, -1);
  const holds = env.relsOf(sources, c.bases).some((r) => holdsProtected(env.contract, r));
  return holds || (c.name === "rm" && env.relsOf(sources, c.tight, true).includes("")) ? MSG.folder : undefined;
}

const DELETERS = new Set(["rm", "mv", "shred", "unlink", "rmdir"]);

/** `find <folder holding protected paths> ... -delete | -exec rm`, and `rsync` into protected paths or `--delete` into a folder holding them. */
function bulkRule(env, c) {
  const { contract, relsOf } = env;
  if (c.name === "find") {
    const start = [];
    for (const a of c.args) {
      if (/^-|^[()!]$/.test(a.text)) break;
      start.push(a);
    }
    const k = c.args.findIndex((a) => /^-(?:exec|execdir|ok|okdir)$/.test(a.text));
    const deletes = c.args.some((a) => a.text === "-delete") || (k >= 0 && DELETERS.has(env.fold(baseName(c.args[k + 1]?.text ?? ""))));
    return deletes && relsOf(start, c.bases).some((r) => isProtectedOrDir(contract, r) || holdsProtected(contract, r)) ? MSG.folder : undefined;
  }
  if (c.name === "rsync") {
    const { flags, operands } = splitArgs(c.args);
    const dest = relsOf(operands.slice(-1), c.bases);
    if (dest.some((r) => isProtectedOrDir(contract, r))) return MSG.protected;
    return flags.some((x) => /^--(?:delete|remove-source-files)/.test(x)) && dest.some((r) => holdsProtected(contract, r)) ? MSG.folder : undefined;
  }
  return undefined;
}

// Whether any of the code texts names a protected path: a protected file, or a protected folder written
// as `folder/` or as a quoted literal.
function mentionsProtected(env, texts, bases) {
  for (const text of texts)
    for (const m of text.matchAll(/[^\s"'`<>|;&()]+/g)) {
      const t = m[0];
      for (const r of env.relsOf([t], bases)) {
        if (isProtected(env.contract, r)) return true;
        if (!isProtectedOrDir(env.contract, r) && !holdsProtected(env.contract, r)) continue;
        if (t.endsWith("/") || ["'", '"', "`"].some((q) => text.includes(q + t + q))) return true;
      }
    }
  return false;
}

/**
 * Interpreter one-liners (`node -e`, `python -c`, `deno eval`) hide the write inside a string; blocked
 * when the code mentions a protected path. Read-only code is blocked too: a read cannot be told from a write.
 */
function oneLinerRule(env, words, bases) {
  const at = words.findIndex((w) => !w.quoted && INTERPRETERS.test(env.fold(baseName(w.text))));
  if (at < 0) return undefined;
  const args = words.slice(at + 1);
  let code;
  if (env.fold(baseName(words[at].text)) === "deno" && args[0]?.text === "eval") code = args.slice(1);
  else
    for (let i = 0; i < args.length; i++) {
      const t = args[i].text;
      if (!t.startsWith("-")) break;
      if (EVAL_FLAG.test(t)) {
        code = args.slice(i + 1);
        break;
      }
      if (/^--?(?:r|require|import|loader|experimental-loader)$/.test(t)) i++;
    }
  return code && mentionsProtected(env, code.map((w) => w.text), bases) ? MSG.oneLiner : undefined;
}

const addAll = (list, items) => {
  for (const x of items) list.push(x);

/**
 * A write command, or a command named by a variable, behind prefixes the parser does not know
 * (`timeout 5 rm x`, `sudo -u root rm x`, `$X x`, `xargs rm` fed by a pipe or heredoc): every word counts as
 * a target. Only known-safe readers are skipped (`grep rm architecture.json`); their redirects are checked apart.
 */
function fallbackRule(env, parsed, idx, writeRels) {
  const { contract, fold: f, relsOf } = env;
  const s = parsed.segments[idx];
  if (knownSafe(s, f)) return undefined;
  const word = (w) => f(baseName(w.text));
  const at = s.words.findIndex((w, k) => {
    if (w.quoted && /\s/.test(w.text)) return false;
    const n = word(w);
    return WRITERS.has(n) || n === "dd" || ((n === "sed" || n === "perl") && inPlace(n, s.words.slice(k + 1).map((x) => x.text)));
  });
  const writer = at < 0 ? undefined : s.words[at];
  const head = s.words.find((w) => !KEYWORDS.has(w.text));
  if (!writer && !head?.text.startsWith("$") && !head?.fromVar) return undefined;
  const wname = writer ? word(writer) : "";
  const after = s.words.slice(at < 0 ? 1 : at + 1);
  const prev = idx > s.pipeStart ? parsed.segments[idx - 1] : undefined;
  const bodyWords = (seg) => seg.bodies.flatMap((b) => b.split(/\s+/).filter(Boolean));
  const fed = s.words.some((w) => !w.quoted && word(w) === "xargs") ? [...(prev ? [...prev.words, ...bodyWords(prev)] : []), ...bodyWords(s)] : [];
  // Any word may be a target (a dd `of=` is unwrapped); only the words after the write command are checked as packages.
  const targets = [...fed, ...s.words].map((w) => (typeof w !== "string" && w.text.startsWith("of=") ? w.text.slice(3) : w));
  const rels = relsOf(targets, s.bases);
  const appends = after.some((w) => /^-[A-Za-z]*a|^--append/.test(w.text));
  if (rels.includes(f("CLAUDE.md")) && (["rm", "mv", "cp", "truncate", "sed", "perl"].includes(wname) || (wname === "tee" && !appends)))
    return MSG.claude;
  if (rels.some((r) => isProtectedOrDir(contract, r))) return MSG.protected;
  // `echo docs | xargs rm -rf`: the words that feed xargs are the folders it removes.
  const fedWords = wname === "rm" && after.some((w) => /^-[A-Za-z]*[rR]|^--recursive$/.test(w.text)) ? fed.map((w) => (typeof w === "string" ? { text: w } : w)) : [];
  const folder = removeRule(env, { name: wname, args: [...fedWords, ...after], bases: s.bases, tight: s.tight });
  if (folder) return folder;
  addAll(writeRels, relsOf(after.filter((w) => !w.text.startsWith("-")), s.bases));
  return undefined;
}

const inPlace = (name, flags) =>
  (name === "sed" && flags.some((x) => /^-[A-Za-z]*i|^--in-place/.test(x))) ||
  (name === "perl" && flags.some((x) => /^-[A-Za-z]*i/.test(x)));

const WRITE_WORD = /\b(?:mkdir|touch|cp|mv|rm|tee|truncate|ln|install|dd|chmod|chown|chgrp|unlink|shred|rmdir)\b|\bsed\s+-i|\bperl\s+-p?i/;

/**
 * Backstop for text that becomes a command at run time (`echo 'rm x' | sh`, `bash <<< '...'`, `bash <(...)`,
 * `bash script.sh`, `eval`, `$X` or `$(...)` as the command, `bash -c --`, `xargs -a`, `node` reading stdin):
 * it cannot be parsed ahead, so the old rough check applies to the whole raw command, quotes included: a write
 * word and a protected path anywhere block it. A folder holding protected paths counts when a recursive delete,
 * `-delete` or `--delete` is named, and with an interpreter its file-changing calls count as write words.
 */
function runtimeTextRule(env, cmd, parsed) {
  const { contract, fold: f } = env;
  const isFlag = (t) => t.startsWith("-");
  const dynamic = (w) => w.dyn || /\$[A-Za-z_{]|^\$$/.test(w.text);
  // Heredoc bodies are data (or checked on their own), so they are left out of the command-position test.
  let bare = cmd;
  for (const seg of parsed.segments) for (const b of seg.bodies) if (b) bare = bare.replace(b, "");
  // [ \t]* rather than \s*: a run of newlines must not be rescanned from every newline.
  let runtime = /(?:^|[;&|(\n])[ \t]*"?(?:\$[({\w]|`)/.test(bare);
  // `strong` is run-time text of unknown content; running a script file is `runtime` but not `strong`.
  let strong = runtime;
  // Scripts under protected paths cannot have been changed by an agent; running one is not run-time text.
  const ownScripts = [];
  let interpreter = false;
  for (const [idx, s] of parsed.segments.entries()) {
    if (s.words.some((w) => !w.quoted && INTERPRETERS.test(f(baseName(w.text))))) interpreter = true;
    if (strong) continue;
    const texts = s.words.map((w) => w.text);
    if (s.words.some((w, k) => !w.quoted && f(baseName(w.text)) === "xargs" && texts.slice(k + 1).some((t) => /^-[A-Za-z]*a$|^--arg-file/.test(t)))) runtime = strong = true;
    const k = s.words.findIndex((w) => !w.quoted && (SHELL_NAME.test(f(baseName(w.text))) || INTERPRETERS.test(f(baseName(w.text))) || [".", "source", "eval"].includes(w.text)));
    if (k < 0) continue;
    const name = f(baseName(texts[k]));
    const args = s.words.slice(k + 1);
    const after = texts.slice(k + 1);
    if (name === "eval") runtime = strong = true;
    else if (SHELL_NAME.test(name) && after.some((t) => /^-[A-Za-z]*c$/.test(t))) {
      // `-c` takes a string; one built at run time (`"$(...)"`, `$X`, `--`, or nothing before a backtick) is run-time text.
      const j = after.findIndex((t) => /^-[A-Za-z]*c$/.test(t));
      if (!args[j + 1] || after[j + 1] === "--" || dynamic(args[j + 1])) runtime = strong = true;
    } else {
      const leading = after.slice(0, after.findIndex((t) => !isFlag(t)) + 1 || after.length);
      const evalAt = name === "deno" && after[0] === "eval" ? 0 : leading.findIndex((t) => EVAL_FLAG.test(t));
      const file = after.find((t) => !isFlag(t));
      if (evalAt >= 0 && (name === "deno" || !/^-.*=/.test(leading[evalAt]))) {
        // An interpreter's `-c` / `-e` string built at run time (`python3 -c "$(echo ...)"`) is run-time text too.
        const code = args[evalAt + 1];
        if (!code || code.dyn || code.text === "$" || (!code.quoted && dynamic(code))) runtime = strong = true;
        continue;
      }
      if (evalAt >= 0) continue;
      if (file === undefined) {
        if (idx > s.pipeStart || s.inputs.length > 0 || s.procSub) runtime = strong = true;
      } else if (env.relsOf([file], s.bases).some((r) => isProtected(contract, r))) ownScripts.push(file);
      else if (!parsed.checked.has(path.normalize(file))) runtime = true;
    }
  }
  if (!runtime) return undefined;
  if (strong && DECODER.test(bare)) return MSG.decode;
  const REDIRECT_TO = />[>|]?[ \t]*["']?([^\s"'<>|;&()\\]+)/g;
  // Adjacent empty quotes are dropped too (`'r''m'` runs rm), so both spellings are checked.
  const squashed = cmd.replace(/(?:''|"")+/g, "");
  for (const text of squashed === cmd ? [cmd] : [cmd, squashed]) {
    const tokens = [...text.matchAll(/[^\s"'<>|;&()\\]+/g)].flatMap((m) => (m[0].startsWith("of=") ? [m[0], m[0].slice(3)] : [m[0]]));
    const targets = [...text.matchAll(REDIRECT_TO)].map((m) => m[1]);
    const protectedIn = (words) => env.relsOf(words, [env.root]).some((r) => isProtectedOrDir(contract, r));
    // A redirect into a protected path is a write on its own; other writes need a write word.
    if (protectedIn(targets)) return MSG.runtime;
    const writes = WRITE_WORD.test(text) || (interpreter && INTERP_WRITE.test(text));
    const folderDelete = FOLDER_DELETE.test(text);
    if (!writes && !folderDelete) continue;
    const seen = tokens.filter((t) => !ownScripts.includes(t));
    if (writes && protectedIn(seen)) return MSG.runtime;
    if (folderDelete && env.relsOf(seen, [env.root]).some((r) => holdsProtected(contract, r))) return MSG.runtime;
  }
  return undefined;
}

function directRule(ctx, cmd, parsed) {
  const { contract, root, f, expand, budget } = ctx;
  if (parsed.tooComplex) return MSG.complex;
  // Root-relative paths of the words, against each base; paths outside the project are ignored.
  // Every lookup spends from the shared budget; once it is gone nothing is resolved and the call is blocked.
  const relsOf = (words, bases, keepRoot = false) => {
    budget.left -= words.length * bases.length;
    if (budget.over || budget.left < 0) {
      budget.over = true;
      return [];
    }
    return words.flatMap((w) => {
      const e = expand(typeof w === "string" ? w : w.text);
      return bases.flatMap((b) => {
        const r = toPosix(path.relative(root, path.resolve(b, e)));
        return (keepRoot || r) && !r.startsWith("..") && !path.isAbsolute(r) ? [r] : [];
      });
    });
  };
  const env = { contract, root, expand, fold: f, relsOf };
  const claudeMd = f("CLAUDE.md");

  const writeRels = [];
  for (const r of parsed.redirects) {
    const rels = relsOf([r.target], r.bases);
    if (r.op === ">" && rels.includes(claudeMd)) return MSG.claude;
    addAll(writeRels, rels);
  }

  for (const [idx, s] of parsed.segments.entries()) {
    const msg =
      gitRule(env, s.words, s.bases, parsed) ??
      oneLinerRule(env, s.words, s.bases) ??
      (mentionsProtected(env, s.codeBodies, s.bases) ? MSG.oneLiner : undefined) ??
      fallbackRule(env, parsed, idx, writeRels);
    if (msg) return msg;
  }
  if (budget.over) return MSG.complex;

  for (const c of parsed.commands) {
    const { flags: fl, operands } = splitArgs(c.args);
    if (WRITERS.has(c.name) || inPlace(c.name, fl)) {
      const rels = relsOf(operands, c.bases);
      const appends = fl.some((x) => /^-[A-Za-z]*a|^--append/.test(x));
      if (rels.includes(claudeMd) && (["rm", "mv", "cp", "truncate", "sed", "perl"].includes(c.name) || (c.name === "tee" && !appends)))
        return MSG.claude;
      addAll(writeRels, rels);
    }
    if (c.name === "dd")
      for (const a of c.args) if (a.text.startsWith("of=")) addAll(writeRels, relsOf([a.text.slice(3)], c.bases));
    // `--output=path` (git diff/log, ...) and `sort -o` / `less -o` write to their target.
    c.args.forEach((a, k) => {
      const long = /^--output(?:=(.*))?$/s.exec(a.text);
      const short = c.name === "sort" || c.name === "less" ? /^-[A-Za-z]*[oO](.*)$/s.exec(a.text) : null;
      const target = long ? (long[1] ?? c.args[k + 1]?.text) : short ? (short[1] || c.args[k + 1]?.text) : undefined;
      if (target) addAll(writeRels, relsOf([target], c.bases));
    });
    const msg = removeRule(env, c) ?? bulkRule(env, c);
    if (msg) return msg;
  }
  const runtimeMsg = runtimeTextRule(env, cmd, parsed);
  if (runtimeMsg) return runtimeMsg;
  if (budget.over) return MSG.complex;
  if (/--no-verify\b/.test(cmd) || /\bgit\s+commit\b[^|;&]*\s-n\b/.test(cmd))
    return "Skipping git hooks is forbidden (--no-verify).";
  if (/core\.hooksPath/.test(cmd) && !/\.githooks/.test(cmd)) return "Changing the git hooks path is forbidden.";
  // pm-guard:begin
  const chosenPm = contract.packageManager;
  if (chosenPm) {
    // Optional command prefixes: sudo/time/env/corepack..., their flags and VAR=value assignments.
    const wrappers =
      "(?:(?:sudo|time|env|command|exec|nohup|nice|corepack)\\s+|-\\S+\\s+|[A-Za-z_]\\w*=\\S*\\s+)*";
    for (const pm of ["pnpm", "npm", "yarn", "bun"].filter((x) => x !== chosenPm)) {
      const install = new RegExp(
        `(^|[;&|(])\\s*${wrappers}${pm}(\\s*($|[;&|)])|\\s+(install|i|add|ci|update|up|upgrade)\\b)`,
      );
      if (cmd.split("\n").some((l) => install.test(l)))
        return `This project uses ${chosenPm} (architecture.json > package_manager); "${pm}" installs would create a second lockfile. Use ${chosenPm} instead.`;
    }
  }
  // pm-guard:end

  // Flags and their values may sit between the PM and the verb (`pnpm --filter x add`, `pnpm -w add`).
  if (/\b(pnpm|npm|yarn|bun)\b[^|;&\n]*\s(add|install|i|update|up)\b/.test(cmd)) {
    const hit = deferredDepInCommand(contract, cmd);
    if (hit) return `"${hit.dep}" cannot be installed. ${deferredHint(hit.deferred)}`;
  }

  if (budget.over) return MSG.complex;
  if (writeRels.some((r) => isProtectedOrDir(contract, r))) return MSG.protected;
  for (const r of writeRels) if (packageOf(contract, r)) {
    const msg = pathRule(contract, r);
    if (msg) return msg;
  }
  return undefined;
}

function bashRule(ctx, cmd, depth = 0, scope = undefined) {
  const parsed = targetsOf(cmd, ctx.root, ctx.expand, ctx.f, scope);
  const own = directRule(ctx, cmd, parsed);
  if (own !== undefined) return own;
  // The same rules apply to every command string another command runs (`bash -c`, `eval`, `$(...)`...).
  const inner = innerStrings(parsed);
  if (depth >= 3) return inner.length ? MSG.nested : undefined;
  for (const item of inner) {
    const msg = bashRule(ctx, typeof item === "string" ? item : item.text, depth + 1, { vars: parsed.vars, budget: parsed.budget });
    if (msg !== undefined) return typeof item === "string" ? msg : `${msg} ${item.hint}`;
  }
  return undefined;
}

/**
 * @param {{ tool: string, input?: any, contract: ReturnType<typeof import("./arch-contract.mjs").contractFrom>,
 *   root: string, home?: string, claudeMdText?: string, platform?: string | undefined }} args
 * @returns {{ allow: true } | { allow: false, message: string }}
 */
export function decide({ tool, input, contract, root, home = "~", claudeMdText, platform }) {
  const ti = input ?? {};
  const ci = CASE_INSENSITIVE.has(platform);
  const f = ci ? lower : (s) => s;
  const expand = (t) => f(t.replace(/^~(?=\/)/, home));
  const ctx = { contract: ci ? foldContract(contract) : contract, root: f(root), f, expand, budget: { left: WORK_BUDGET, over: false } };
  let message;
  if (["Write", "Edit", "MultiEdit", "NotebookEdit"].includes(tool))
    message = fileRule(ctx, tool, ti, claudeMdText);
  else if (tool === "Bash") message = bashRule(ctx, String(ti.command ?? ""));
  return message === undefined ? { allow: true } : { allow: false, message };
}

const READ_ONLY_GIT = new Set(["log", "grep", "show", "diff", "status", "blame", "ls-files", "describe", "rev-parse"]);

export const baseName = (w) => w.replace(/^.*[\\/]/, "");

const MAX_LEX_LEVEL = 6;
const ANSI_SIMPLE = { n: "\n", t: "\t", r: "\r", a: "\x07", b: "\b", e: "\x1b", E: "\x1b", f: "\f", v: "\v", "\\": "\\", "'": "'", '"': '"', "?": "?" };

// The `$'...'` string whose body starts at i: escapes are skipped (so `\'` does not end it) and decoded.
// Returns the text and the index of the closing quote (or the string length).
function ansiC(s, i) {
  let text = "";
  for (; i < s.length && s[i] !== "'"; i++) {
    if (s[i] !== "\\") {
      text += s[i];
      continue;
    }
    const n = s[++i];
    if (n === undefined) break;
    if (Object.hasOwn(ANSI_SIMPLE, n)) text += ANSI_SIMPLE[n];
    else if (n === "x" || n === "u" || n === "U") {
      const max = n === "x" ? 2 : n === "u" ? 4 : 8;
      const m = new RegExp(`^[0-9a-fA-F]{1,${max}}`).exec(s.slice(i + 1, i + 1 + max));
      const code = m ? Number.parseInt(m[0], 16) : Number.NaN;
      if (m && code <= 0x10ffff) text += String.fromCodePoint(code);
      else text += `\\${n}${m ? m[0] : ""}`;
      i += m ? m[0].length : 0;
    } else if (/[0-7]/.test(n)) {
      const m = /^[0-7]{1,3}/.exec(s.slice(i, i + 3));
      text += String.fromCharCode(Number.parseInt(m[0], 8) & 255);
      i += m[0].length - 1;
    } else if (n === "c") text += String.fromCharCode((s.charCodeAt(++i) || 0) & 31);
    else text += `\\${n}`;
  }
  return { text, end: i };
}

// `<<[-]WORD` starting just after the `<<`: the delimiter and where the word ends.
function readHeredocDelimiter(s, i) {
  let j = i;
  const strip = s[j] === "-";
  if (strip) j++;
  while (s[j] === " " || s[j] === "\t") j++;
  let delim = "";
  let quoted = false;
  for (; j < s.length && !/[\s;|&()<>]/.test(s[j]); j++) {
    const c = s[j];
    if (c === "$" && (s[j + 1] === "'" || s[j + 1] === '"')) continue;
    if (c === "'" || c === '"') {
      quoted = true;
      const e = s.indexOf(c, j + 1);
      const stop = e < 0 ? s.length : e;
      delim += s.slice(j + 1, stop);
      j = stop;
    } else if (c === "\\") {
      quoted = true;
      delim += s[++j] ?? "";
    } else delim += c;
  }
  return { delim, quoted, strip, end: j, body: "" };
}

// The body starting at `from` (just after a newline) up to its delimiter line; `next` is the index of
// the newline that ends the delimiter line (or the string length).
function heredocBody(s, from, h) {
  for (let pos = from; pos <= s.length; ) {
    const e = s.indexOf("\n", pos);
    const stop = e < 0 ? s.length : e;
    const line = s.slice(pos, stop).replace(/\r$/, "");
    if ((h.strip ? line.replace(/^\t+/, "") : line) === h.delim) {
      h.body = s.slice(from, pos);
      return { next: stop };
    }
    if (e < 0) break;
    pos = e + 1;
  }
  h.body = s.slice(from);
  return { next: s.length };
}

// Index of the first `)` of the `))` that closes an arithmetic expression starting at i.
function closeArith(s, i) {
  let depth = 0;
  for (; i < s.length; i++) {
    if (s[i] === "(") depth++;
    else if (s[i] === ")") {
      if (depth === 0 && s[i + 1] === ")") return i;
      depth = Math.max(0, depth - 1);
    }
  }
  return s.length;
}

// Index of the `)` closing the `(` at i. This runs the same lexer as shellSegments, so comments,
// quotes, arithmetic and heredocs are recognised by one set of rules.
function closeParen(s, i, state, level) {
  if (level >= MAX_LEX_LEVEL) {
    state.complex = true;
    return s.length;
  }
  return lexShell(s, state, i + 1, true, level + 1).end;
}

// `$(...)` and backtick substitutions in text the shell expands (unquoted heredoc bodies).
function substitutionsIn(text, out, state, level) {
  for (let j = 0; j < text.length; j++) {
    if (text[j] === "\\") j++;
    else if (text[j] === "$" && text[j + 1] === "(") {
      const e = closeParen(text, j + 1, state, level);
      out.push(text.slice(j + 2, e));
      j = e;
    } else if (text[j] === "`") {
      let e = j + 1;
      while (e < text.length && text[e] !== "`") e += text[e] === "\\" ? 2 : 1;
      out.push(text.slice(j + 1, e));
      j = e;
    }
  }
}

/**
 * The one shell lexer. Splits a command into simple commands: quotes are removed, `;` `&&` `||` `|` `&`
 * newlines, parentheses and backticks separate commands, comments (`#` at a word start) are dropped and
 * redirects (`>`, `>>`, `>|`, `>&file`, `&>`) are collected apart from the words. `$(...)`, backticks and
 * arithmetic inside double quotes or arithmetic are returned in `nested` as command strings of their own.
 * Each segment records the operator that ended it (`sep`) and its heredocs: `<<WORD` is recognised
 * outside quotes, comments and arithmetic, and the body is skipped rather than parsed as commands.
 * With `stopAtClose` it stops at the `)` that closes an enclosing `$(` and returns its index in `end`.
 */
function lexShell(cmd, state, from = 0, stopAtClose = false, level = 0) {
  const segments = [];
  const nested = [];
  let words = [];
  let redirs = [];
  let heredocs = [];
  let inputs = [];
  let procSub = false;
  let waiting = [];
  let cur = null;
  let pending = null;
  let depth = 0;
  const pushWord = () => {
    if (cur === null) return;
    const w = cur;
    cur = null;
    if (pending) {
      const p = pending;
      pending = null;
      if (p.op === "<" || p.op === "<<<") inputs.push({ op: p.op, target: w.text });
      else if (!(p.amp && /^(\d+|-)$/.test(w.text))) redirs.push({ op: p.op, target: w.text });
    } else words.push(w);
  };
  const end = (sep) => {
    pushWord();
    pending = null;
    if (words.length || redirs.length || inputs.length || procSub) segments.push({ words, redirs, sep, heredocs, inputs, procSub });
    words = [];
    redirs = [];
    heredocs = [];
    inputs = [];
    procSub = false;
  };
  const grow = () => {
    cur ??= { text: "", quoted: false };
    return cur;
  };
  // Arithmetic (`$((..))`, `$[..]`) and backticks are scanned for commands; returns the new index.
  const expansion = (i) => {
    grow().dyn = true;
    if (cmd[i + 2] === "(") {
      const e = closeArith(cmd, i + 3);
      nested.push(cmd.slice(i + 3, e));
      return e + 1;
    }
    const e = cmd.indexOf("]", i + 2);
    const stop = e < 0 ? cmd.length : e;
    nested.push(cmd.slice(i + 2, stop));
    return stop;
  };
  let i = from;
  for (; i < cmd.length; i++) {
    const c = cmd[i];
    if (c === "\\") {
      if (cmd[i + 1] === "\n") i++;
      else if (i + 1 < cmd.length) grow().text += cmd[++i];
    } else if (c === "$" && cmd[i + 1] === "'") {
      const r = ansiC(cmd, i + 2);
      grow().quoted = true;
      cur.text += r.text;
      i = r.end;
    } else if (c === "$" && cmd[i + 1] === '"') {
      // $"..." quotes like "...": drop the $.
    } else if (c === "$" && ((cmd[i + 1] === "(" && cmd[i + 2] === "(") || cmd[i + 1] === "[")) {
      i = expansion(i);
      grow().text += "0";
    } else if (c === "'") {
      const e = cmd.indexOf("'", i + 1);
      const stop = e < 0 ? cmd.length : e;
      grow().quoted = true;
      cur.text += cmd.slice(i + 1, stop);
      i = stop;
    } else if (c === '"') {
      grow().quoted = true;
      for (i++; i < cmd.length && cmd[i] !== '"'; i++) {
        if (cmd[i] === "\\" && '"\\$`'.includes(cmd[i + 1] ?? "")) i++;
        else if (cmd[i] === "$" && ((cmd[i + 1] === "(" && cmd[i + 2] === "(") || cmd[i + 1] === "[")) {
          i = expansion(i);
          continue;
        } else if (cmd[i] === "$" && cmd[i + 1] === "(") {
          const e = closeParen(cmd, i + 1, state, level);
          nested.push(cmd.slice(i + 2, e));
          cur.dyn = true;
          i = e;
          continue;
        } else if (cmd[i] === "`") {
          let e = i + 1;
          while (e < cmd.length && cmd[e] !== "`") e += cmd[e] === "\\" ? 2 : 1;
          nested.push(cmd.slice(i + 1, e));
          cur.dyn = true;
          i = e;
          continue;
        }
        cur.text += cmd[i];
      }
    } else if (c === " " || c === "\t" || c === "\r") pushWord();
    else if (c === "#" && cur === null) {
      const e = cmd.indexOf("\n", i);
      i = e < 0 ? cmd.length : e - 1;
    } else if (c === "|") {
      const sep = cmd[i + 1] === "|" ? "||" : "|";
      end(sep);
      if (sep === "||") i++;
    } else if (c === "<" && cmd[i + 1] === "<" && cmd[i + 2] !== "<") {
      if (cur !== null && !cur.quoted && /^\d+$/.test(cur.text)) cur = null;
      else pushWord();
      const h = readHeredocDelimiter(cmd, i + 2);
      if (h.delim) {
        heredocs.push(h);
        waiting.push(h);
      }
      i = h.end - 1;
    } else if (c === "\n" && waiting.length) {
      for (const h of waiting) {
        i = heredocBody(cmd, i + 1, h).next;
        // An unquoted delimiter lets the shell expand substitutions inside the body.
        if (!h.quoted) substitutionsIn(h.body, nested, state, level);
      }
      waiting = [];
      end(";");
    } else if (c === "(" && cmd[i + 1] === "(" && cur === null && words.every((w) => KEYWORDS.has(w.text))) {
      // `(( expr ))` as a command: arithmetic, not two subshells.
      const e = closeArith(cmd, i + 2);
      nested.push(cmd.slice(i + 2, e));
      i = e + 1;
    } else if (c === "(") {
      if (cur?.text.endsWith("$")) cur.dyn = true;
      depth++;
      end(";");
    } else if (c === ")") {
      if (stopAtClose && depth === 0) break;
      depth = Math.max(0, depth - 1);
      end(";");
    } else if (c === "`") {
      // An unquoted backtick ends the word but leaves its value to run time (`X=`cmd``).
      if (cur !== null) cur.dyn = true;
      end(";");
    } else if (c === "\n" || c === ";") end(";");
    else if (c === "&" && cmd[i + 1] !== ">") {
      const sep = cmd[i + 1] === "&" ? "&&" : "&";
      end(sep);
      if (sep === "&&") i++;
    } else if (c === "<" && cmd[i + 1] === "(") {
      // `<(...)`: process substitution, a command's output read as a file.
      pushWord();
      procSub = true;
    } else if (c === ">" || c === "<" || c === "&") {
      if (cur !== null && !cur.quoted && /^\d+$/.test(cur.text)) cur = null;
      else pushWord();
      if (c === "&") i++;
      let op = c === "<" ? "<" : ">";
      let amp = false;
      if (c === "<") {
        let n = 1;
        while (cmd[i + 1] === "<") {
          i++;
          n++;
        }
        if (n >= 3) op = "<<<";
      } else if (cmd[i + 1] === ">") {
        op = ">>";
        i++;
      } else if (cmd[i + 1] === "|") i++;
      else if (cmd[i + 1] === "&") {
        amp = true;
        i++;
      }
      pending = { op, amp };
    } else grow().text += c;
  }
  end("");
  return { segments, nested, end: i };
}

const shellSegments = (cmd) => {
  const state = { complex: false };
  const { segments, nested } = lexShell(cmd, state);
  return { segments, nested, complex: state.complex };
};

/** Drops assignments, keywords and wrapper prefixes (and their flags); `find -exec` yields its inner command too. */
function commandsOf(words, fold) {
  let i = 0;
  while (i < words.length) {
    const w = words[i].text;
    if (ASSIGNMENT.test(w) || KEYWORDS.has(w)) i++;
    else if (WRAPPERS.has(fold(baseName(w)))) {
      i++;
      while (i < words.length && words[i].text.startsWith("-")) i++;
    } else break;
  }
  if (i >= words.length) return [];
  const name = fold(baseName(words[i].text));
  const args = words.slice(i + 1);
  const out = [{ name, args }];
  if (name === "find") {
    const k = args.findIndex((a) => /^-(?:exec|execdir|ok|okdir)$/.test(a.text));
    if (k >= 0) out.push(...commandsOf(args.slice(k + 1), fold));
  }
  return out;
}

/** Splits arguments into flags and operands; everything after `--` is an operand. */
export function splitArgs(args) {
  const flags = [];
  const operands = [];
  let rest = false;
  for (const a of args) {
    if (!rest && a.text === "--") rest = true;
    else if (!rest && a.text.length > 1 && a.text.startsWith("-")) flags.push(a.text);
    else operands.push(a);
  }
  return { flags, operands };
}

// git global options and the position of the subcommand.
export function gitOptions(args) {
  let i = 0;
  const dirs = [];
  for (; i < args.length; i++) {
    const t = args[i].text;
    if (t === "-C" || t === "-c" || /^--(?:git-dir|work-tree|namespace|config-env)$/.test(t)) {
      if (t === "-C" && args[i + 1]) dirs.push(args[i + 1].text);
      i++;
    } else if (!t.startsWith("-")) break;
  }
  return { i, dirs };
}

// Files a segment writes: its `>` / `>>` redirect targets and the operands of tee.
function writtenTargets(seg, fold) {
  const out = seg.redirs.filter((r) => r.op === ">" || r.op === ">>").map((r) => path.normalize(r.target));
  const at = seg.words.findIndex((w) => !w.quoted && fold(baseName(w.text)) === "tee");
  if (at >= 0) for (const w of seg.words.slice(at + 1)) if (!w.text.startsWith("-")) out.push(path.normalize(w.text));
  return out;
}

// Whether a segment executes one of the files: `bash f`, `. f`, `node f`, or `./f` itself.
function runs(seg, targets, fold) {
  const is = (w) => targets.includes(path.normalize(w.text));
  if (seg.words[0] && is(seg.words[0])) return true;
  return seg.words.some(
    (w, k) =>
      !w.quoted &&
      (SHELL_NAME.test(fold(baseName(w.text))) || w.text === "." || w.text === "source" || INTERPRETERS.test(fold(baseName(w.text)))) &&
      seg.words.slice(k + 1).some(is),
  );
}

/** `git commit -F -` and `gh ... --body-file -` read their input as data. */
function readsStdinAsData(seg) {
  if (seg.name !== "git" && seg.name !== "gh") return false;
  const w = seg.words.map((x) => x.text);
  return w.some((t, k) => (/^(?:-F|--file|--body-file|--notes-file)$/.test(t) && w[k + 1] === "-") || /^--(?:file|body-file|notes-file)=-$/.test(t));
}

/**
 * A read-only tool run plainly: no variable assignment (`PAGER=rm`), no git `-c`, and none of the
 * options that make a reader run or write something (`rg --pre`, `man -P`, `--ext-diff`, `--output`, `-o`).
 */
export function knownSafe(seg, fold) {
  const name = seg.name ?? "";
  if (seg.words.some((w) => ASSIGNMENT.test(w.text))) return false;
  if (seg.words.some((w) => /^--(?:pre|pre-glob|pager|ext-diff|output|exec-path|upload-pack)(?:=|$)/.test(w.text))) return false;
  if (name === "man" && seg.words.some((w) => /^-[A-Za-z]*P/.test(w.text))) return false;
  if ((name === "less" || name === "more") && seg.words.some((w) => /^-[A-Za-z]*[oO]/.test(w.text))) return false;
  if (name === "git") {
    const at = seg.words.findIndex((w) => !w.quoted && fold(baseName(w.text)) === "git");
    const args = seg.words.slice(at + 1);
    const { i } = gitOptions(args);
    return !args.slice(0, i).some((a) => a.text === "-c") && READ_ONLY_GIT.has(fold(args[i]?.text ?? ""));
  }
  return READ_ONLY.has(name);
}

const VAR_REF = /\$(?:\{([A-Za-z_]\w*)\}|([A-Za-z_]\w*))/g;
const IFS_REF = /\$(?:\{IFS\}|IFS(?!\w))(?:\$[0-9])?/g;
const MAX_VAR_LENGTH = 1024;
// Total characters variable substitution may produce in one decision, shared by nested strings.
const MAX_EXPANSION = 65_536;
const ASSIGNING = new Set(["export", "declare", "typeset", "readonly", "local"]);
const ASSIGN_WORD = /^([A-Za-z_]\w*)(\+?)=/;
// Commands that set the variables named by their bare arguments.
const SETTERS = new Set(["read", "getopts", "mapfile", "readarray", "unset", "printf", "select", "for"]);

/** `$NAME` / `${NAME}` replaced by the value assigned earlier in the line; `$IFS` / `${IFS}` become a space. */
function expandVars(text, vars, budget) {
  const spend = (v) => {
    budget.left -= v.length;
    return budget.left < 0 ? "" : v;
  };
  return text.replace(IFS_REF, () => spend(" ")).replace(VAR_REF, (m, a, b) => {
    const v = vars.get(a ?? b);
    return v === undefined ? m : spend(v);
  });
}

/**
 * Records `NAME=value` segments (also behind `export`) in `vars` and substitutes known variables into the
 * words and redirect targets of every other segment, splitting a word that expanded to several. A value
 * built at run time (`$(...)`, backticks), appended to (`+=`) or set by `read`, `getopts`, `for`... is
 * forgotten, so `$NAME` stays unexpanded and the rules for variables of unknown value apply.
 * Returns false once the expansion budget is spent.
 */
function applyVars(seg, vars, budget) {
  const first = seg.words[0]?.text;
  const rest = first !== undefined && ASSIGNING.has(first) ? seg.words.slice(1) : seg.words;
  if (rest.length && rest.every((w) => ASSIGN_WORD.test(w.text))) {
    for (const w of rest) {
      const m = ASSIGN_WORD.exec(w.text);
      const value = expandVars(w.text.slice(m[0].length), vars, budget);
      if (w.dyn || (m[2] && !vars.has(m[1]))) vars.delete(m[1]);
      else vars.set(m[1], ((m[2] ? vars.get(m[1]) : "") + value).slice(0, MAX_VAR_LENGTH));
    }
    return budget.left >= 0;
  }
  const at = seg.words.findIndex((w) => !ASSIGN_WORD.test(w.text) && !KEYWORDS.has(w.text));
  if (at >= 0 && SETTERS.has(seg.words[at].text))
    for (const w of seg.words.slice(at + 1)) if (/^[A-Za-z_]\w*$/.test(w.text)) vars.delete(w.text);
  seg.words = seg.words.flatMap((w) => {
    if (!w.text.includes("$")) return [w];
    const text = expandVars(w.text, vars, budget);
    if (text === w.text || budget.left < 0) return [w];
    const parts = text.split(/\s+/).filter(Boolean);
    return (parts.length ? parts : [""]).map((t) => ({ ...w, text: t, fromVar: true }));
  });
  for (const r of seg.redirs) r.target = expandVars(r.target, vars, budget);
  for (const r of seg.inputs) r.target = expandVars(r.target, vars, budget);
  return budget.left >= 0;
}

/**
 * The commands and redirects of a shell command line, each with the directories it may run in.
 * `bases` are absolute and grow linearly: the project root, the chain of every `cd` / `pushd` target
 * assuming each succeeds, and every target resolved against the root alone (a `cd` can fail).
 * `tight` is the directory after the latest `cd` chained with `&&` or `;` (a `cd x || cmd` keeps it).
 * `tooComplex` is set past MAX_CDS directory changes or MAX_COMMAND_LENGTH characters.
 * @param {string} cmd
 * @param {string} root
 * @param {(t: string) => string} [expand] applied to every path word (home, case folding)
 * @param {(t: string) => string} [fold] applied to command names
 * @param {{ vars: Map<string, string>, budget: { left: number } }} [outer] variables assigned before this string
 *   runs (`X=a; bash -c 'rm $X'`) and the substitution budget shared with the strings around it
 */
export function targetsOf(cmd, root, expand = (t) => t, fold = (t) => t, outer = { vars: new Map(), budget: { left: MAX_EXPANSION } }) {
  const vars = new Map(outer.vars);
  const { budget } = outer;
  const commands = [];
  const redirects = [];
  const segments = [];
  if (cmd.length > MAX_COMMAND_LENGTH || budget.left < 0) return { commands, redirects, segments, nested: [], vars, budget, tooComplex: true };
  const parsed = shellSegments(cmd);
  if (parsed.complex) return { commands, redirects, segments, nested: [], vars, budget, tooComplex: true };
  let chain = root;
  let bases = [root];
  let tight = [root];
  let cds = 0;
  let pipeStart = 0;
  for (const seg of parsed.segments) {
    if (!applyVars(seg, vars, budget)) return { commands, redirects, segments, nested: parsed.nested, vars, budget, tooComplex: true };
    for (const r of seg.redirs) redirects.push({ op: r.op, target: r.target, bases });
    const found = commandsOf(seg.words, fold);
    const bodies = seg.heredocs.map((h) => h.body);
    segments.push({ words: seg.words, redirs: seg.redirs, inputs: seg.inputs, procSub: seg.procSub, pipeStart, bases, tight, name: found[0]?.name, sep: seg.sep, bodies });
    if (seg.sep !== "|") pipeStart = segments.length;
    for (const c of found) {
      commands.push({ ...c, bases, tight });
      if (c.name !== "cd" && c.name !== "pushd") continue;
      const dir = splitArgs(c.args).operands[0];
      if (!dir || dir.text === "-") continue;
      if (++cds > MAX_CDS) return { commands, redirects, segments, nested: parsed.nested, vars, budget, tooComplex: true };
      chain = path.resolve(chain, expand(dir.text));
      bases = [...new Set([...bases, chain, path.resolve(root, expand(dir.text))])];
      if (seg.sep === "&&" || seg.sep === ";") tight = [chain];
    }
  }
  const checked = new Set();
  for (const [idx, seg] of segments.entries()) {
    seg.scriptBodies = [];
    seg.codeBodies = [];
    if (!seg.bodies.length) continue;
    // A heredoc body is data only for a known sink: cat/tee/read-only tools (piped only into read-only
    // tools or tee), `git commit -F -`, `gh ... --body-file -`. Any other consumer runs it as a script.
    const pipe = [];
    for (let j = idx; segments[j]?.sep === "|" && segments[j + 1]; j++) pipe.push(segments[j + 1]);
    const sink =
      (knownSafe(seg, fold) || seg.name === "tee" || readsStdinAsData(seg)) &&
      pipe.every((x) => knownSafe(x, fold) || x.name === "tee");
    if (sink) {
      // A sink that writes a file which a later command in the same line runs: the body is that script.
      const targets = writtenTargets(seg, fold);
      const runner = targets.length ? segments.slice(idx + 1).find((x) => runs(x, targets, fold)) : undefined;
      if (!runner) continue;
      for (const t of targets) checked.add(t);
      seg.scriptBodies = seg.bodies;
      if (runner.words.some((w) => !w.quoted && INTERPRETERS.test(fold(baseName(w.text))))) seg.codeBodies = seg.bodies;
      continue;
    }
    seg.scriptBodies = seg.bodies;

import { createHash, randomBytes } from "node:crypto";
import { basename, join } from "node:path";
import { parseEnv } from "node:util";

/** `<slug>_<sha1(path)[:6]>`; slug = lowercased basename, `[^a-z0-9]` → `_`, max 20 chars. */
export function worktreeKey(worktreePath) {
  const slug = basename(worktreePath)
    .toLowerCase()
    .replace(/[^a-z0-9]/g, "_")
    .slice(0, 20);
  const hash = createHash("sha1").update(worktreePath).digest("hex").slice(0, 6);
  return `${slug}_${hash}`;
}

/** Separates the project prefix from the worktree key in every worktree database name. */
export const WORKTREE_SEPARATOR = "_wt_";

/** Database name for the main checkout (`key === null`) or a worktree. */
export function databaseName(prefix, key) {
  return key === null ? prefix : `${prefix}${WORKTREE_SEPARATOR}${key}`;
}

/** True only for a database named `<prefix>_wt_<something>`; a literal comparison, no wildcards. */
export function isWorktreeDatabase(prefix, name) {
  const head = `${prefix}${WORKTREE_SEPARATOR}`;
  return typeof name === "string" && name.length > head.length && name.startsWith(head);
}

/** SQL LIKE pattern (default `\` escape) matching worktree databases of `prefix`; `\`, `%`, `_` are literal. */
export function likePatternFor(prefix) {
  const literal = `${prefix}${WORKTREE_SEPARATOR}`.replace(/[\\%_]/g, "\\$&");
  return `${literal}%`;
}

/** Ordered `.env` sources: Paseo source checkout, main checkout, then this checkout's `.env.example`. */
export function sourceEnvCandidates({ paseoSource, mainCheckout, toplevel }) {
  return [
    ...(paseoSource ? [join(paseoSource, ".env")] : []),
    join(mainCheckout, ".env"),
    join(toplevel, ".env.example"),
  ];
}

/**
 * Databases that must never be dropped: the bare prefix plus each env's DATABASE_URL database.
 * `envs` is `[{ source, text }]`. A malformed DATABASE_URL throws (naming its source) so callers
 * fail closed instead of dropping a database they could not identify.
 */
export function protectedDatabases({ prefix, envs }) {
  const names = new Set([prefix]);
  for (const { source, text } of envs) {
    const url = parseEnv(text).DATABASE_URL;
    if (!url) continue;
    try {
      names.add(databaseOf(url));
    } catch {
      throw new Error(`DATABASE_URL in ${source} is not a valid URL; refusing to drop anything`);
    }
  }
  return names;
}

/** Drop-safety rule: only this project's worktree databases that are not protected. */
export function mayDropDatabase({ prefix, database, protected: protectedNames }) {
  return isWorktreeDatabase(prefix, database) && !protectedNames.has(database);
}

/** `url` with its database name replaced; `undefined` passes through. */
export function withDatabase(url, database) {
  if (url === undefined || url === "") return undefined;
  const parsed = new URL(url);
  parsed.pathname = `/${database}`;
  return parsed.toString();
}

/** Database name from a connection URL. */
export function databaseOf(url) {
  return decodeURIComponent(new URL(url).pathname.slice(1));
}

export function setLine(text, key, value) {
  const line = new RegExp(`^${key}=.*$`, "m");
  if (line.test(text)) return text.replace(line, () => `${key}=${value}`);
  const base = text === "" || text.endsWith("\n") ? text : `${text}\n`;
  return `${base}${key}=${value}\n`;
}

/**
 * Points DATABASE_URL and WORKTREE_KEY at this checkout. A key is rewritten only while it is
 * missing or still equals the source checkout's value, so deliberate edits survive re-runs.
 */
export function rewriteIsolatedKeys(current, source, database, key) {
  const now = parseEnv(current);
  const src = parseEnv(source);
  const targets = {
    DATABASE_URL: withDatabase(now.DATABASE_URL ?? src.DATABASE_URL, database),
    WORKTREE_KEY: key ?? "",
  };
  let text = current;
  for (const [name, value] of Object.entries(targets)) {
    if (value === undefined) continue;
    if (now[name] === undefined || now[name] === src[name]) text = setLine(text, name, value);
  }
  return text;
}

/** Fills each listed key that is empty or still a `change-me` placeholder with 32 random bytes (hex). */
export function ensureSecrets(text, names) {
  const env = parseEnv(text);
  let result = text;
  for (const name of names) {
    const value = (env[name] ?? "").trim();
    if (value === "" || value.startsWith("change-me")) {
      result = setLine(result, name, randomBytes(32).toString("hex"));
    }
  }
  return result;
}

/** mkdir-based lock in the shared git dir so concurrent setups (Paseo + hooks) do not race. */
async function withLock(fn) {
  const lock = join(commonDir, "worktree-setup.lock");
  for (let attempt = 0; ; attempt++) {
    try {
      mkdirSync(lock);
      break;
    } catch {
      const stale = existsSync(lock) && Date.now() - statSync(lock).mtimeMs > 10 * 60_000;
      if (stale) rmSync(lock, { recursive: true, force: true });
      else if (attempt > 120) throw new Error(`could not acquire ${lock}`);
      else await new Promise((resolve) => setTimeout(resolve, 500));
    }
  }
  try {
    return await fn();
  } finally {
    rmSync(lock, { recursive: true, force: true });
  }

async function gc(confirmed) {
  const env = readEnv();
  const protectedDbs = protectedSourceDatabases();
  const sql = adminSql(env);
  try {
    const rows = await sql`
      select datname, shobj_description(oid, 'pg_database') as comment
      from pg_database where datname like ${likePatternFor(PREFIX)}`;
    const orphans = rows.filter(
      (r) =>
        mayDropDatabase({ prefix: PREFIX, database: r.datname, protected: protectedDbs }) &&
        r.comment?.startsWith(COMMENT_PREFIX) &&
